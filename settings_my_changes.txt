  Future<void> _showAddCurrencyDialog() async {
    final currencyController = TextEditingController();

    final currency = await Get.dialog<String>(
      AlertDialog(
        title: Text('Add Currency'.tr),
        content: TextField(
          controller: currencyController,
          autofocus: true,
          textCapitalization: TextCapitalization.characters,
          maxLength: 10,
          decoration: InputDecoration(
            labelText: 'Currency code or symbol'.tr,
            hintText: 'e.g. AED'.tr,
          ),
          onSubmitted: (value) {
            final code = value.trim();
            if (code.isNotEmpty) {
              Get.back(result: code);
            }
          },
        ),
        actions: [
          TextButton(
            onPressed: () => Get.back(),
            child: Text('Cancel'.tr),
          ),
          FilledButton(
            onPressed: () {
              final code = currencyController.text.trim();
              if (code.isNotEmpty) {
                Get.back(result: code);
              }
            },
            child: Text('Add'.tr),
          ),
        ],
      ),
    );

    currencyController.dispose();

    if (currency != null && currency.trim().isNotEmpty) {
      await controller.addCustomCurrency(currency);
    }
  }

  Future<void> _showLanguageListDialog(BuildContext context) async {
    final isDark = Theme.of(context).brightness == Brightness.dark;

    await Get.dialog<void>(
      AlertDialog(
        backgroundColor: isDark ? AppColors.cardDark : AppColors.surfaceLight,
        surfaceTintColor: Colors.transparent,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(26)),
        title: Text(
          'Select Language'.tr,
          style: AppTextStyles.headingMedium(
            isDark,
          ).copyWith(fontSize: 19, fontWeight: FontWeight.w700),
        ),
        content: SizedBox(
          width: 400,
          child: ConstrainedBox(
            constraints: const BoxConstraints(maxHeight: 390),
            child: SingleChildScrollView(
              child: Obx(
                () => Column(
                  mainAxisSize: MainAxisSize.min,
                  children: OnboardingController.languages.map((language) {
                    final code = language.code.toLowerCase();
                    final isSelected =
                        controller.selectedLanguage.value == code;

                    return ListTile(
                      leading: Icon(
                        isSelected
                            ? Icons.radio_button_checked_rounded
                            : Icons.radio_button_off_rounded,
                        color: isSelected
                            ? AppColors.primary
                            : AppColors.textSecondary(isDark),
                      ),
                      title: Text(language.nativeName),
                      subtitle: Text(language.englishName),
                      onTap: () async {
                        await controller.changeLanguage(code);
                        Get.back<void>();
                      },
                    );
                  }).toList(),
                ),
              ),
            ),
          ),
        ),
        actions: [
          OutlinedButton(
            onPressed: () => Get.back<void>(),
            child: Text('Close'.tr),
          ),
        ],
      ),
    );
  }

  Future<void> _confirmDeleteCurrency(
    BuildContext context,
    String currency,
  ) async {
    final confirmed = await Get.dialog<bool>(
      AlertDialog(
        title: Text('Delete Currency?'.tr),
        content: Text(
          'Are you sure you want to remove "@currency" from your custom currencies?'
              .trParams({'currency': currency}),
        ),
        actions: [
          TextButton(
            onPressed: () => Get.back(result: false),
            child: Text('Cancel'.tr),
          ),
          TextButton(
            onPressed: () => Get.back(result: true),
            child: Text(
              'Delete'.tr,
              style: const TextStyle(color: AppColors.expenseRed),
            ),
          ),
        ],
      ),
    );

    if (confirmed == true) {
      await controller.deleteCustomCurrency(currency);
    }
  }
